     1→"use client";
     2→import { useCallback, useEffect, useRef, useState } from "react";
     3→import { motion, AnimatePresence } from "framer-motion";
     4→import { X } from "lucide-react";
     5→import gsap from "gsap";
     6→import { ScrollTrigger } from "gsap/ScrollTrigger";
     7→import { CountUp } from "@/components/sections/_shared";
     8→import {
     9→  METRO_STATIONS,
    10→  METRO_INTRO,
    11→  type MetroStation,
    12→  type StrategyItem,
    13→} from "@/lib/data";
    14→import { useSound } from "@/hooks/use-sound";
    15→import { usePrefersReducedMotion } from "@/hooks/use-prefers-reduced-motion";
    16→import { useBodyScrollLock } from "@/hooks/use-body-scroll-lock";
    17→import { useFocusTrap } from "@/hooks/use-focus-trap";
    18→import { getLenis } from "@/lib/lenis-instance";
    19→
    20→const EASE = [0.16, 1, 0.3, 1] as const;
    21→
    22→/* ============================================================
    23→   SECTION 04 — BEST WORK / DELHI METRO
    24→   Gamified horizontal metro track with pinned GSAP scroll,
    25→   step-out deep-dive overlays, keyboard nav, Hindi announcement
    26→   ticker, door-chime SFX. Degrades to a vertical stacked grid
    27→   below 1024px or when prefers-reduced-motion is set.
    28→
    29→   Station content (per spec Section 06):
    30→   - Each station now carries headline + StrategyItem[] + extras.
    31→   - The deep-dive overlay exposes the full case study:
    32→     headline, problem, strategy as numbered steps, impact,
    33→     all metrics as CountUp grid, plus extras as labelled lists.
    34→   ============================================================ */
    35→export default function BestWorkMetro() {
    36→  const { play } = useSound();
    37→  const reduced = usePrefersReducedMotion();
    38→
    39→  const [isDesktop, setIsDesktop] = useState(false);
    40→  const [activeIndex, setActiveIndex] = useState(0);
    41→  const [openStationId, setOpenStationId] = useState<string | null>(null);
    42→  const [inView, setInView] = useState(false);
    43→  const [showKeyHint, setShowKeyHint] = useState(false);
    44→
    45→  const sectionRef = useRef<HTMLElement>(null);
    46→  const outerRef = useRef<HTMLDivElement>(null);
    47→  const viewportRef = useRef<HTMLDivElement>(null);
    48→  const trackRef = useRef<HTMLDivElement>(null);
    49→  const trainRef = useRef<HTMLDivElement>(null);
    50→
    51→  // Latest active index + play ref so the GSAP onUpdate closure can read
    52→  // fresh values without re-subscribing on every render.
    53→  const progressRef = useRef(0);
    54→  const activeRef = useRef(0);
    55→  const playRef = useRef(play);
    56→
    57→  // Track the trigger button so focus can be restored when the deep-dive
    58→  // overlay closes.
    59→  const triggerRef = useRef<HTMLElement | null>(null);
    60→  const deepDivePanelRef = useRef<HTMLDivElement | null>(null);
    61→
    62→  // Keep playRef in sync with the latest play closure without touching refs
    63→  // during render (which would trip react-hooks/refs).
    64→  useEffect(() => {
    65→    playRef.current = play;
    66→  }, [play]);
    67→
    68→  const showPinned = isDesktop && !reduced;
    69→
    70→  const openStation =
    71→    METRO_STATIONS.find((s) => s.id === openStationId) ?? null;
    72→  const nextStation =
    73→    METRO_STATIONS[(activeIndex + 1) % METRO_STATIONS.length];
    74→
    75→  // Lock scroll + trap focus while the deep-dive overlay is open.
    76→  useBodyScrollLock(openStationId !== null);
    77→  useFocusTrap(deepDivePanelRef, openStationId !== null, triggerRef);
    78→
    79→  /* ---- desktop breakpoint detection ---- */
    80→  useEffect(() => {
    81→    const mq = window.matchMedia("(min-width: 1024px)");
    82→    const update = () => setIsDesktop(mq.matches);
    83→    update();
    84→    mq.addEventListener("change", update);
    85→    return () => mq.removeEventListener("change", update);
    86→  }, []);
    87→
    88→  /* ---- section-in-view tracking for keyboard nav ---- */
    89→  // Uses a generous rootMargin so keyboard nav activates as soon as the
    90→  // metro section is approaching the viewport, not just when it's 15% visible.
    91→  useEffect(() => {
    92→    const section = sectionRef.current;
    93→    if (!section) return;
    94→    const io = new IntersectionObserver(
    95→      ([entry]) => {
    96→        setInView(entry.isIntersecting);
    97→        if (entry.isIntersecting) {
    98→          // Show the prominent keyboard hint the first time the user enters
    99→          setShowKeyHint(true);
   100→          // Auto-dismiss after 5s (the user will have seen it by then)
   101→          setTimeout(() => setShowKeyHint(false), 5000);
   102→        }
   103→      },
   104→      { rootMargin: "-10% 0px -10% 0px", threshold: 0 }
   105→    );
   106→    io.observe(section);
   107→    return () => io.disconnect();
   108→  }, []);
   109→
   110→  /* ---- GSAP pinned horizontal track (desktop + not reduced) ---- */
   111→  useEffect(() => {
   112→    if (!showPinned) return;
   113→    const outer = outerRef.current;
   114→    const viewport = viewportRef.current;
   115→    const track = trackRef.current;
   116→    if (!outer || !viewport || !track) return;
   117→
   118→    gsap.registerPlugin(ScrollTrigger);
   119→
   120→    const ctx = gsap.context(() => {
   121→      const getDistance = () =>
   122→        Math.max(0, track.scrollWidth - window.innerWidth);
   123→
   124→      gsap.to(track, {
   125→        x: () => -getDistance(),
   126→        ease: "none",
   127→        scrollTrigger: {
   128→          trigger: outer,
   129→          start: "top top",
   130→          end: () => "+=" + (getDistance() + window.innerHeight),
   131→          scrub: 1,
   132→          pin: viewport,
   133→          anticipatePin: 1,
   134→          invalidateOnRefresh: true,
   135→          onUpdate: (self) => {
   136→            progressRef.current = self.progress;
   137→            const idx = Math.min(
   138→              METRO_STATIONS.length - 1,
   139→              Math.floor(self.progress * METRO_STATIONS.length)
   140→            );
   141→            // State guard: only fire door chime + state update when the
   142→            // active index actually changes. Prevents repeated sound firing
   143→            // during minor scrub reversals.
   144→            if (idx !== activeRef.current) {
   145→              activeRef.current = idx;
   146→              setActiveIndex(idx);
   147→              playRef.current("door");
   148→            }
   149→            // Train marker ● travels along the blue line with progress.
   150→            if (trainRef.current) {
   151→              const tx = self.progress * (window.innerWidth - 40);
   152→              trainRef.current.style.transform = `translate3d(${tx}px, -50%, 0)`;
   153→            }
   154→          },
   155→          onLeaveBack: () => {
   156→            // Reset to the first station when scrolling back above the section.
   157→            if (activeRef.current !== 0) {
   158→              activeRef.current = 0;
   159→              setActiveIndex(0);
   160→            }
   161→            if (trainRef.current) {
   162→              trainRef.current.style.transform = "translate3d(0, -50%, 0)";
   163→            }
   164→          },
   165→        },
   166→      });
   167→
   168→      ScrollTrigger.refresh();
   169→    }, outer);
   170→
   171→    const onResize = () => ScrollTrigger.refresh();
   172→    window.addEventListener("resize", onResize);
   173→
   174→    return () => {
   175→      window.removeEventListener("resize", onResize);
   176→      ctx.revert();
   177→    };
   178→  }, [showPinned]);
   179→
   180→  /* ---- keyboard nav (←/→) active only when section in view + pinned ---- */
   181→  const scrollToStation = useCallback((idx: number) => {
   182→    const outer = outerRef.current;
   183→    const track = trackRef.current;
   184→    if (!outer || !track) return;
   185→    const outerTop = outer.getBoundingClientRect().top + window.scrollY;
   186→    const distance = Math.max(0, track.scrollWidth - window.innerWidth);
   187→    const totalScroll = distance + window.innerHeight;
   188→    const stationScroll = totalScroll * (idx / METRO_STATIONS.length);
   189→    const targetTop = outerTop + stationScroll;
   190→    const lenis = getLenis();
   191→    if (lenis) {
   192→      lenis.scrollTo(targetTop, { duration: 1.2 });
   193→    } else {
   194→      window.scrollTo({ top: targetTop, behavior: "smooth" });
   195→    }
   196→  }, []);
   197→
   198→  useEffect(() => {
   199→    if (!inView || !showPinned || openStationId) return;
   200→    const onArrow = (e: Event) => {
   201→      const dir = (e as CustomEvent<"left" | "right">).detail;
   202→      if (dir !== "right" && dir !== "left") return;
   203→      // Dismiss the keyboard hint — the user got the message.
   204→      setShowKeyHint(false);
   205→      const delta = dir === "right" ? 1 : -1;
   206→      const target = Math.max(
   207→        0,
   208→        Math.min(METRO_STATIONS.length - 1, activeRef.current + delta)
   209→      );
   210→      if (target === activeRef.current) return;
   211→      playRef.current("blip");
   212→      scrollToStation(target);
   213→    };
   214→    const onHome = () => {
   215→      if (activeRef.current === 0) return;
   216→      playRef.current("blip");
   217→      scrollToStation(0);
   218→    };
   219→    const onEnd = () => {
   220→      if (activeRef.current === METRO_STATIONS.length - 1) return;
   221→      playRef.current("blip");
   222→      scrollToStation(METRO_STATIONS.length - 1);
   223→    };
   224→    window.addEventListener("baaz:arrow", onArrow as EventListener);
   225→    window.addEventListener("baaz:metro-home", onHome);
   226→    window.addEventListener("baaz:metro-end", onEnd);
   227→    return () => {
   228→      window.removeEventListener("baaz:arrow", onArrow as EventListener);
   229→      window.removeEventListener("baaz:metro-home", onHome);
   230→      window.removeEventListener("baaz:metro-end", onEnd);
   231→    };
   232→  }, [inView, showPinned, openStationId, scrollToStation]);
   233→
   234→  /* ---- Esc closes the deep-dive overlay ---- */
   235→  useEffect(() => {
   236→    if (!openStationId) return;
   237→    const onKey = (e: KeyboardEvent) => {
   238→      if (e.key === "Escape") {
   239→        setOpenStationId(null);
   240→        playRef.current("confirm");
   241→      }
   242→    };
   243→    window.addEventListener("keydown", onKey);
   244→    return () => window.removeEventListener("keydown", onKey);
   245→  }, [openStationId]);
   246→
   247→  /* ---- handlers ---- */
   248→  const enterMetro = () => {
   249→    playRef.current("door");
   250→    const outer = outerRef.current;
   251→    if (!outer) return;
   252→    const lenis = getLenis();
   253→    if (lenis) {
   254→      lenis.scrollTo(outer, { duration: 1.2 });
   255→    } else {
   256→      outer.scrollIntoView({ behavior: "smooth", block: "start" });
   257→    }
   258→  };
   259→
   260→  const openDeepDive = (e: React.MouseEvent<HTMLElement>, station: MetroStation) => {
   261→    triggerRef.current = e.currentTarget;
   262→    setOpenStationId(station.id);
   263→    play("door");
   264→  };
   265→
   266→  const closeDeepDive = () => {
   267→    setOpenStationId(null);
   268→    play("confirm");
   269→  };
   270→
   271→  const returnToTop = () => {
   272→    const sec = sectionRef.current;
   273→    if (!sec) return;
   274→    const lenis = getLenis();
   275→    if (lenis) {
   276→      lenis.scrollTo(sec, { duration: 1.2 });
   277→    } else {
   278→      sec.scrollIntoView({ behavior: "smooth", block: "start" });
   279→    }
   280→  };
   281→
   282→  // Footer parts split on "·" so we can render the "Return to Platform"
   283→  // segment as a button.
   284→  const footerParts = METRO_INTRO.footer
   285→    .split("·")
   286→    .map((s) => s.trim())
   287→    .filter(Boolean);
   288→
   289→  return (
   290→    <section
   291→      ref={sectionRef}
   292→      id="best-work"
   293→      className="env-black relative w-full"
   294→      aria-labelledby="best-work-header"
   295→      data-cursor-label="best work"
   296→    >
   297→      {/* CSS keyframes for the Hindi ticker marquee */}
   298→      <style>{`
   299→        @keyframes baazMetroTicker {
   300→          from { transform: translate3d(0, 0, 0); }
   301→          to   { transform: translate3d(-50%, 0, 0); }
   302→        }
   303→      `}</style>
   304→
   305→      {/* ====================================================
   306→          A. INTRO PANEL
   307→          ==================================================== */}
   308→      <div className="mx-auto w-full max-w-[1200px] px-5 py-24 sm:px-8 sm:py-32 lg:px-12">
   309→        {/* Index header — mirrors SectionShell */}
   310→        <motion.div
   311→          className="mb-10 flex items-baseline gap-3 border-b border-white/10 pb-3 font-mono text-[11px] uppercase tracking-widest text-[#6B6B6B] sm:mb-14"
   312→          initial={{ opacity: 0, y: 14 }}
   313→          whileInView={{ opacity: 1, y: 0 }}
   314→          viewport={{ once: true, margin: "-10% 0px" }}
   315→          transition={{ duration: 0.6, ease: EASE }}
   316→        >
   317→          <span className="text-[#FFD400]">04</span>
   318→          <span id="best-work-header" className="text-[#F4F1EA]/70">
   319→            BEST WORK / DELHI METRO
   320→          </span>
   321→          <span className="ml-auto hidden h-px flex-1 bg-white/10 sm:block" />
   322→          <span className="hidden sm:inline">{"// baaz.sys"}</span>
   323→        </motion.div>
   324→
   325→        {/* Metro logo mark + blinking "Next train: NOW" */}
   326→        <div className="mb-10 flex flex-wrap items-center gap-4">
   327→          <div
   328→            className="flex items-center gap-3"
   329→            data-cursor-label={METRO_INTRO.line}
   330→          >
   331→            <span className="flex h-10 w-10 items-center justify-center rounded-full border-2 border-[#FFD400] bg-[#FFD400] text-[#0A0A0A]">
   332→              <span className="font-mono text-sm font-bold">M</span>
   333→            </span>
   334→            <span className="font-mono text-xs uppercase tracking-[0.3em] text-[#FFD400]">
   335→              {METRO_INTRO.line}
   336→            </span>
   337→          </div>
   338→          <span className="ml-auto flex items-center gap-2 font-mono text-[11px] uppercase tracking-[0.25em] text-[#F4F1EA]/70">
   339→            <span className="inline-block h-2 w-2 rounded-full bg-[#FFD400] blink" />
   340→            Next train: NOW
   341→          </span>
   342→        </div>
   343→
   344→        {/* Hindi heading */}
   345→        <motion.h2
   346→          className="font-deva text-4xl font-bold leading-tight text-[#F4F1EA] sm:text-6xl"
   347→          initial={{ opacity: 0, y: 24 }}
   348→          whileInView={{ opacity: 1, y: 0 }}
   349→          viewport={{ once: true, margin: "-10% 0px" }}
   350→          transition={{ duration: 0.8, ease: EASE }}
   351→          data-cursor-label={METRO_INTRO.english}
   352→        >
   353→          {METRO_INTRO.hindi}
   354→        </motion.h2>
   355→
   356→        {/* English subtitle */}
   357→        <motion.p
   358→          className="mt-4 font-display text-2xl font-bold tracking-tight text-[#F4F1EA]/60 sm:text-3xl"
   359→          initial={{ opacity: 0, y: 24 }}
   360→          whileInView={{ opacity: 1, y: 0 }}
   361→          viewport={{ once: true, margin: "-10% 0px" }}
   362→          transition={{ duration: 0.8, delay: 0.15, ease: EASE }}
   363→        >
   364→          {METRO_INTRO.english}
   365→        </motion.p>
   366→
   367→        {/* CTA + sub copy */}
   368→        <motion.div
   369→          className="mt-12 flex flex-wrap items-center gap-6"
   370→          initial={{ opacity: 0, y: 24 }}
   371→          whileInView={{ opacity: 1, y: 0 }}
   372→          viewport={{ once: true, margin: "-10% 0px" }}
   373→          transition={{ duration: 0.8, delay: 0.3, ease: EASE }}
   374→        >
   375→          <button
   376→            type="button"
   377→            onClick={enterMetro}
   378→            onMouseEnter={() => play("tick")}
   379→            data-cursor-label="enter metro"
   380→            className="group inline-flex items-center gap-3 border-2 border-[#FFD400] bg-[#FFD400] px-6 py-3.5 font-mono text-xs uppercase tracking-[0.3em] text-[#0A0A0A] transition-colors hover:bg-transparent hover:text-[#FFD400] focus-ring"
   381→          >
   382→            <span className="h-2 w-2 rounded-full bg-[#0A0A0A] transition-colors group-hover:bg-[#FFD400]" />
   383→            {METRO_INTRO.cta}
   384→            <span aria-hidden>→</span>
   385→          </button>
   386→
   387→          <p className="max-w-md font-mono text-xs uppercase tracking-[0.18em] text-[#6B6B6B]">
   388→            {`// ${METRO_STATIONS.length} stations · ${METRO_STATIONS.length} case studies · scroll to ride`}
   389→          </p>
   390→        </motion.div>
   391→      </div>
   392→
   393→      {/* ====================================================
   394→          B. PINNED HORIZONTAL TRACK (desktop) — OR STACKED (mobile/reduced)
   395→          ==================================================== */}
   396→      {showPinned ? (
   397→        <div ref={outerRef} className="relative w-full">
   398→          <div
   399→            ref={viewportRef}
   400→            className="relative h-screen w-full overflow-hidden bg-[#0A0A0A]"
   401→          >
   402→            {/* Yellow horizontal BLUE LINE through the middle */}
   403→            <div
   404→              aria-hidden
   405→              className="pointer-events-none absolute left-0 right-0 top-1/2 z-0 h-px -translate-y-1/2 bg-[#FFD400]/70"
   406→            />
   407→            {/* Line endcaps */}
   408→            <span
   409→              aria-hidden
   410→              className="pointer-events-none absolute left-0 top-1/2 z-0 h-3 w-1 -translate-y-1/2 bg-[#FFD400]"
   411→            />
   412→            <span
   413→              aria-hidden
   414→              className="pointer-events-none absolute right-0 top-1/2 z-0 h-3 w-1 -translate-y-1/2 bg-[#FFD400]"
   415→            />
   416→
   417→            {/* Train marker ● — travels along the blue line with scroll progress */}
   418→            <div
   419→              ref={trainRef}
   420→              aria-hidden
   421→              className="pointer-events-none absolute left-4 top-1/2 z-30 h-5 w-5 -translate-y-1/2 rounded-full bg-[#FFD400]"
   422→              style={{
   423→                transform: "translate3d(0, -50%, 0)",
   424→                boxShadow: "0 0 18px rgba(255,212,0,0.55)",
   425→              }}
   426→            />
   427→
   428→            {/* Top status bar */}
   429→            <div className="absolute left-0 right-0 top-0 z-20 flex items-center justify-between gap-4 border-b border-white/10 bg-[#0A0A0A]/80 px-5 py-3 backdrop-blur-sm sm:px-8">
   430→              <div className="flex items-center gap-3 font-mono text-[10px] uppercase tracking-[0.25em]">
   431→                <span className="inline-block h-2 w-2 rounded-full bg-[#FFD400] blink" />
   432→                <span className="text-[#FFD400]">{"● BLUE LINE"}</span>
   433→                <span className="text-[#6B6B6B]">/</span>
   434→                <span className="text-[#F4F1EA]/80">DELHI METRO</span>
   435→              </div>
   436→              <div className="flex items-center gap-3 font-mono text-[10px] uppercase tracking-[0.25em]">
   437→                {/* Persistent keyboard hint — animated key icons */}
   438→                <span className="flex items-center gap-1.5">
   439→                  <kbd className="inline-flex h-5 w-5 items-center justify-center border border-[#FFD400]/50 bg-[#FFD400]/10 font-mono text-[10px] text-[#FFD400] animate-pulse">←</kbd>
   440→                  <kbd className="inline-flex h-5 w-5 items-center justify-center border border-[#FFD400]/50 bg-[#FFD400]/10 font-mono text-[10px] text-[#FFD400] animate-pulse">→</kbd>
   441→                  <span className="hidden text-[#F4F1EA]/70 sm:inline">navigate</span>
   442→                </span>
   443→                <span className="hidden text-[#6B6B6B] sm:inline">/</span>
   444→                <span className="text-[#F4F1EA]/70 tabular-nums">
   445→                  {String(activeIndex + 1).padStart(2, "0")} / 0
   446→                  {METRO_STATIONS.length}
   447→                </span>
   448→              </div>
   449→            </div>
   450→
   451→            {/* PROMINENT KEYBOARD HINT OVERLAY — appears the first time the
   452→                user enters the metro section, then auto-dismisses. Makes it
   453→                clear that arrow keys navigate the slowly-sliding stations. */}
   454→            <AnimatePresence>
   455→              {showKeyHint && inView && showPinned && !openStationId && (
   456→                <motion.div
   457→                  className="pointer-events-none absolute inset-0 z-40 flex items-center justify-center"
   458→                  initial={{ opacity: 0 }}
   459→                  animate={{ opacity: 1 }}
   460→                  exit={{ opacity: 0 }}
   461→                  transition={{ duration: 0.4 }}
   462→                  aria-hidden
   463→                >
   464→                  <div className="flex flex-col items-center gap-4 rounded-lg border border-[#FFD400]/40 bg-[#0A0A0A]/90 px-8 py-6 backdrop-blur-md">
   465→                    <p className="font-mono text-[10px] uppercase tracking-[0.3em] text-[#FFD400]">
   466→                      {"// KEYBOARD REQUIRED"}
   467→                    </p>
   468→                    <div className="flex items-center gap-3">
   469→                      <motion.div
   470→                        className="flex h-14 w-14 items-center justify-center rounded-lg border-2 border-[#FFD400] bg-[#FFD400]/10 font-mono text-2xl text-[#FFD400]"
   471→                        animate={{ scale: [1, 1.15, 1], opacity: [0.7, 1, 0.7] }}
   472→                        transition={{ duration: 1.2, repeat: Infinity, ease: "easeInOut" }}
   473→                      >
   474→                        ←
   475→                      </motion.div>
   476→                      <motion.div
   477→                        className="flex h-14 w-14 items-center justify-center rounded-lg border-2 border-[#FFD400] bg-[#FFD400]/10 font-mono text-2xl text-[#FFD400]"
   478→                        animate={{ scale: [1, 1.15, 1], opacity: [0.7, 1, 0.7] }}
   479→                        transition={{ duration: 1.2, repeat: Infinity, ease: "easeInOut", delay: 0.6 }}
   480→                      >
   481→                        →
   482→                      </motion.div>
   483→                    </div>
   484→                    <p className="text-center font-display text-lg font-bold tracking-tight text-[#F4F1EA]">
   485→                      USE ARROW KEYS
   486→                    </p>
   487→                    <p className="text-center font-mono text-[10px] uppercase tracking-[0.2em] text-[#6B6B6B]">
   488→                      to navigate between stations
   489→                    </p>
   490→                  </div>
   491→                </motion.div>
   492→              )}
   493→            </AnimatePresence>
   494→
   495→            {/* Route-map mini-indicator — all 5 stations as dots with
