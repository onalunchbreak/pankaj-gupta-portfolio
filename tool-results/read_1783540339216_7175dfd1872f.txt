     9→import { useSound } from "@/hooks/use-sound";

    11→import { getLenis } from "@/lib/lenis-instance";
    12→
    13→const EASE = [0.16, 1, 0.3, 1] as const;
    14→
    15→/* ============================================================

    24→  const reduced = usePrefersReducedMotion();
    25→
    26→  const [isDesktop, setIsDesktop] = useState(false);
    27→  const [activeIndex, setActiveIndex] = useState(0);
    28→  const [openStationId, setOpenStationId] = useState<string | null>(null);
    29→  const [inView, setInView] = useState(false);
    30→
    31→  const sectionRef = useRef<HTMLElement>(null);
    32→  const outerRef = useRef<HTMLDivElement>(null);
    33→  const viewportRef = useRef<HTMLDivElement>(null);
    34→  const trackRef = useRef<HTMLDivElement>(null);
    35→  const trainRef = useRef<HTMLDivElement>(null);
    36→
    37→  // Latest active index + play ref so the GSAP onUpdate closure can read
    38→  // fresh values without re-subscribing on every render.
    39→  const progressRef = useRef(0);
    40→  const activeRef = useRef(0);
    41→  const playRef = useRef(play);
    42→
    43→  // Keep playRef in sync with the latest play closure without touching refs
    44→  // during render (which would trip react-hooks/refs).
    45→  useEffect(() => {
    46→    playRef.current = play;
    47→  }, [play]);
    48→
    49→  const showPinned = isDesktop && !reduced;
    50→
    51→  const openStation =
    52→    METRO_STATIONS.find((s) => s.id === openStationId) ?? null;
    53→  const nextStation =
    54→    METRO_STATIONS[(activeIndex + 1) % METRO_STATIONS.length];
    55→
    56→  /* ---- desktop breakpoint detection ---- */
    57→  useEffect(() => {
    58→    const mq = window.matchMedia("(min-width: 1024px)");
    59→    const update = () => setIsDesktop(mq.matches);
    60→    update();
    61→    mq.addEventListener("change", update);
    62→    return () => mq.removeEventListener("change", update);
    63→  }, []);
    64→
    65→  /* ---- section-in-view tracking for keyboard nav ---- */
    66→  useEffect(() => {
    67→    const section = sectionRef.current;
    68→    if (!section) return;
    69→    const io = new IntersectionObserver(
    70→      ([entry]) => setInView(entry.isIntersecting),
    71→      { threshold: 0.15 }
    72→    );
    73→    io.observe(section);
    74→    return () => io.disconnect();
    75→  }, []);
    76→
    77→  /* ---- GSAP pinned horizontal track (desktop + not reduced) ---- */
    78→  useEffect(() => {
    79→    if (!showPinned) return;
    80→    const outer = outerRef.current;
    81→    const viewport = viewportRef.current;
    82→    const track = trackRef.current;
    83→    if (!outer || !viewport || !track) return;
    84→
    85→    gsap.registerPlugin(ScrollTrigger);
    86→
    87→    const ctx = gsap.context(() => {
    88→      const getDistance = () =>
    89→        Math.max(0, track.scrollWidth - window.innerWidth);
    90→
    91→      gsap.to(track, {
    92→        x: () => -getDistance(),
    93→        ease: "none",
    94→        scrollTrigger: {
    95→          trigger: outer,
    96→          start: "top top",
    97→          end: () => "+=" + (getDistance() + window.innerHeight),
    98→          scrub: 1,
    99→          pin: viewport,
   100→          anticipatePin: 1,
   101→          invalidateOnRefresh: true,
   102→          onUpdate: (self) => {
   103→            progressRef.current = self.progress;
   104→            const idx = Math.min(
   105→              METRO_STATIONS.length - 1,
   106→              Math.floor(self.progress * METRO_STATIONS.length)
   107→            );
   108→            if (idx !== activeRef.current) {
   109→              activeRef.current = idx;
   110→              setActiveIndex(idx);
   111→              playRef.current("door");
   112→            }
   113→            // Train marker ● travels along the blue line with progress.
   114→            if (trainRef.current) {
   115→              const tx = self.progress * (window.innerWidth - 40);
   116→              trainRef.current.style.transform = `translate3d(${tx}px, -50%, 0)`;
   117→            }
   118→          },
   119→          onLeaveBack: () => {
   120→            // Reset to the first station when scrolling back above the section.
   121→            if (activeRef.current !== 0) {
   122→              activeRef.current = 0;
   123→              setActiveIndex(0);
   124→            }
   125→            if (trainRef.current) {
   126→              trainRef.current.style.transform = "translate3d(0, -50%, 0)";
   127→            }
   128→          },
   129→        },
   130→      });
   131→
   132→      ScrollTrigger.refresh();
   133→    }, outer);
   134→
   135→    const onResize = () => ScrollTrigger.refresh();
   136→    window.addEventListener("resize", onResize);
   137→
   138→    return () => {
   139→      window.removeEventListener("resize", onResize);
   140→      ctx.revert();
   141→    };
   142→  }, [showPinned]);
   143→
   144→  /* ---- keyboard nav (←/→) active only when section in view + pinned ---- */

   146→    const outer = outerRef.current;
   147→    const track = trackRef.current;
   148→    if (!outer || !track) return;
   149→    const outerTop = outer.getBoundingClientRect().top + window.scrollY;
   150→    const distance = Math.max(0, track.scrollWidth - window.innerWidth);
   151→    const totalScroll = distance + window.innerHeight;
   152→    const stationScroll = totalScroll * (idx / METRO_STATIONS.length);
   153→    const targetTop = outerTop + stationScroll;
   154→    const lenis = getLenis();
   155→    if (lenis) {
   156→      lenis.scrollTo(targetTop, { duration: 1.2 });
   157→    } else {
   158→      window.scrollTo({ top: targetTop, behavior: "smooth" });
   159→    }
   160→  }, []);
   161→
   162→  useEffect(() => {
   163→    if (!inView || !showPinned || openStationId) return;

   165→      const dir = (e as CustomEvent<"left" | "right">).detail;
   166→      if (dir !== "right" && dir !== "left") return;
   167→      const delta = dir === "right" ? 1 : -1;
   168→      const target = Math.max(
   169→        0,
   170→        Math.min(METRO_STATIONS.length - 1, activeRef.current + delta)
   171→      );
   172→      if (target === activeRef.current) return;
   173→      playRef.current("blip");
   174→      scrollToStation(target);
   175→    };
   176→    const onHome = () => {
   177→      if (activeRef.current === 0) return;
   178→      playRef.current("blip");
   179→      scrollToStation(0);
   180→    };
   181→    const onEnd = () => {
   182→      if (activeRef.current === METRO_STATIONS.length - 1) return;
   183→      playRef.current("blip");
   184→      scrollToStation(METRO_STATIONS.length - 1);
   185→    };

   187→    window.addEventListener("baaz:metro-home", onHome);
   188→    window.addEventListener("baaz:metro-end", onEnd);
   189→    return () => {
   190→      window.removeEventListener("baaz:arrow", onArrow as EventListener);
   191→      window.removeEventListener("baaz:metro-home", onHome);
   192→      window.removeEventListener("baaz:metro-end", onEnd);
   193→    };
   194→  }, [inView, showPinned, openStationId, scrollToStation]);

   200→      if (e.key === "Escape") {
   201→        setOpenStationId(null);
   202→        playRef.current("confirm");
   203→      }
   204→    };
   205→    window.addEventListener("keydown", onKey);
   206→    return () => window.removeEventListener("keydown", onKey);
   207→  }, [openStationId]);

   239→      sec.scrollIntoView({ behavior: "smooth", block: "start" });
   240→    }
   241→  };
   242→
   243→  // Footer parts split on "·" so we can render the "Return to Platform"

   247→    .map((s) => s.trim())

   250→  return (
   251→    <section
   252→      ref={sectionRef}
   253→      id="best-work"
   254→      className="relative w-full bg-[#0A0A0A]"
   255→    >
   256→      {/* CSS keyframes for the Hindi ticker marquee */}
   257→      <style>{`
   258→        @keyframes baazMetroTicker {
   259→          from { transform: translate3d(0, 0, 0); }

   261→        }
   262→      `}</style>
   263→
   264→      {/* ====================================================
   265→          A. INTRO PANEL
   266→          ==================================================== */}
   267→      <div className="mx-auto w-full max-w-[1200px] px-5 py-24 sm:px-8 sm:py-32 lg:px-12">

   270→          className="mb-10 flex items-baseline gap-3 border-b border-white/10 pb-3 font-mono text-[11px] uppercase tracking-widest text-[#6B6B6B] sm:mb-14"
   271→          initial={{ opacity: 0, y: 14 }}

   279→          <span className="hidden sm:inline">{"// baaz.sys"}</span>

   281→
   282→        {/* Metro logo mark + blinking "Next train: NOW" */}

   285→            <span className="flex h-10 w-10 items-center justify-center rounded-full border-2 border-[#FFD400] bg-[#FFD400] text-[#0A0A0A]">
   286→              <span className="font-mono text-sm font-bold">M</span>
   287→            </span>
   288→            <span className="font-mono text-xs uppercase tracking-[0.3em] text-[#FFD400]">
   289→              {METRO_INTRO.line}
   290→            </span>
   291→          </div>
   292→          <span className="ml-auto flex items-center gap-2 font-mono text-[11px] uppercase tracking-[0.25em] text-[#F4F1EA]/70">
   293→            <span className="inline-block h-2 w-2 rounded-full bg-[#FFD400] blink" />
   294→            Next train: NOW
   295→          </span>
   296→        </div>
   297→
   298→        {/* Hindi heading */}
   299→        <motion.h2
   300→          className="font-deva text-4xl font-bold leading-tight text-[#F4F1EA] sm:text-6xl"
   301→          initial={{ opacity: 0, y: 24 }}
   302→          whileInView={{ opacity: 1, y: 0 }}
   303→          viewport={{ once: true, margin: "-10% 0px" }}
   304→          transition={{ duration: 0.8, ease: EASE }}
   305→          data-cursor-label={METRO_INTRO.english}
   306→        >
   307→          {METRO_INTRO.hindi}
   308→        </motion.h2>
   309→
   310→        {/* English subtitle */}

   321→        {/* CTA + sub copy */}

   335→          >

   343→          </p>
   344→        </motion.div>
   345→      </div>
   346→
   347→      {/* ====================================================
   348→          B. PINNED HORIZONTAL TRACK (desktop) — OR STACKED (mobile/reduced)
   349→          ==================================================== */}
   350→      {showPinned ? (
   351→        <div ref={outerRef} className="relative w-full">
   352→          <div
   353→            ref={viewportRef}
   354→            className="relative h-screen w-full overflow-hidden bg-[#0A0A0A]"
   355→          >
   356→            {/* Yellow horizontal BLUE LINE through the middle */}
   357→            <div
   358→              aria-hidden
   359→              className="pointer-events-none absolute left-0 right-0 top-1/2 z-0 h-px -translate-y-1/2 bg-[#FFD400]/70"
   360→            />
   361→            {/* Line endcaps */}
   362→            <span
   363→              aria-hidden
   364→              className="pointer-events-none absolute left-0 top-1/2 z-0 h-3 w-1 -translate-y-1/2 bg-[#FFD400]"
   365→            />
   366→            <span
   367→              aria-hidden

   371→            {/* Train marker ● — travels along the blue line with scroll progress */}
   372→            <div
   373→              ref={trainRef}
   374→              aria-hidden
   375→              className="pointer-events-none absolute left-4 top-1/2 z-30 h-5 w-5 -translate-y-1/2 rounded-full bg-[#FFD400]"
   376→              style={{
   377→                transform: "translate3d(0, -50%, 0)",
   378→                boxShadow: "0 0 18px rgba(255,212,0,0.55)",
   379→              }}
   380→            />
   381→
   382→            {/* Top status bar */}
   383→            <div className="absolute left-0 right-0 top-0 z-20 flex items-center justify-between gap-4 border-b border-white/10 bg-[#0A0A0A]/80 px-5 py-3 backdrop-blur-sm sm:px-8">
   384→              <div className="flex items-center gap-3 font-mono text-[10px] uppercase tracking-[0.25em]">
   385→                <span className="inline-block h-2 w-2 rounded-full bg-[#FFD400] blink" />
   386→                <span className="text-[#FFD400]">{"● BLUE LINE"}</span>

   392→                  ← → to navigate

   394→                <span className="hidden text-[#6B6B6B] sm:inline">/</span>

   401→
   402→            {/* Route-map mini-indicator — all 5 stations as dots with

   404→            <div className="absolute left-0 right-0 top-[46px] z-20 hidden items-center gap-2 border-b border-white/5 bg-[#0A0A0A]/60 px-5 py-2 backdrop-blur-sm sm:flex sm:px-8">
   405→              <span className="mr-1 font-mono text-[9px] uppercase tracking-[0.3em] text-[#6B6B6B]">
